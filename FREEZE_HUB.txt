----------------------------------------------------------- local function applySnowEffect(parentFrame) 	local running = true 	local flakes = {} 	local function spawnFlake() 		if not (parentFrame and parentFrame.Parent) then return end 		local flake = Instance.new("TextLabel") 		flake.BackgroundTransparency = 1 		flake.Text = "·" 		flake.TextColor3 = Theme.Moonlight 		flake.Font = FontBody 		flake.TextSize = math.random(8, 14) 		flake.TextTransparency = 1 		flake.ZIndex = (parentFrame.ZIndex or 1) + 1 		flake.Size = UDim2.fromOffset(16, 16) 		flake.Parent = parentFrame 		local baseX = math.random() 		local fallSpeed = 0.04 + math.random() * 0.05 -- slow: ~0.04–0.09 scale/sec 		local swayAmp = 0.008 + math.random() * 0.012 		local swaySpeed = 0.35 + math.random() * 0.4 		local phase = math.random() * math.pi * 2 		local spawnTime = os.clock() 		local targetAlpha = 0.35 + math.random() * 0.35 		flake.Position = UDim2.new(baseX, 0, -0.05, 0) 		local conn 		conn = RunService.RenderStepped:Connect(function(dt) 			if not (running and parentFrame and parentFrame.Parent and flake.Parent) then 				conn:Disconnect() 				if flake then flake:Destroy() end 				return 			end 			local t = os.clock() - spawnTime 			local y = -0.05 + t * fallSpeed 			local x = baseX + math.sin(t * swaySpeed + phase) * swayAmp 			flake.Position = UDim2.new(x, 0, y, 0) 			-- soft fade in / out 			local progress = math.clamp((y + 0.05) / 1.15, 0, 1) 			local fadeIn = math.clamp(progress / 0.12, 0, 1) 			local fadeOut = math.clamp((1 - progress) / 0.15, 0, 1) 			flake.TextTransparency = 1 - (targetAlpha * math.min(fadeIn, fadeOut)) 			if y > 1.1 then 				conn:Disconnect() 				flake:Destroy() 			end 		end) 		table.insert(flakes, conn) 	end 	task.spawn(function() 		while running and parentFrame and parentFrame.Parent do 			spawnFlake() 			-- realistic sparse rate: one flake every 0.55–0.95s 			task.wait(0.55 + math.random() * 0.4) 		end 	end) 	return function() 		running = false 		for _, c in ipairs(flakes) do 			pcall(function() c:Disconnect() end) 		end 	end end ----------------------------------------------------------- -- DRAGGABLE ----------------------------------------------------------- local function makeDraggable(frame, handle) 	handle = handle or frame 	local dragging, dragInput, dragStart, startPos 	handle.InputBegan:Connect(function(input) 		if input.UserInputType == Enum.UserInputType.MouseButton1 			or input.UserInputType == Enum.UserInputType.Touch then 			dragging = true 			dragStart = input.Position 			startPos = frame.Position 			input.Changed:Connect(function() 				if input.UserInputState == Enum.UserInputState.End then 					dragging = false 				end 			end) 		end 	end) 	handle.InputChanged:Connect(function(input) 		if input.UserInputType == Enum.UserInputType.MouseMovement 			or input.UserInputType == Enum.UserInputType.Touch then 			dragInput = input 		end 	end) 	UserInputService.InputChanged:Connect(function(input) 		if input == dragInput and dragging then 			local d = input.Position - dragStart 			frame.Position = UDim2.new( 				startPos.X.Scale, startPos.X.Offset + d.X, 				startPos.Y.Scale, startPos.Y.Offset + d.Y 			) 		end 	end) end ----------------------------------------------------------- -- SMOOTH TOGGLE
